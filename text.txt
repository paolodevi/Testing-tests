adams.adm

ADAMS/
TITLE = Single CG body, gravity from user subroutine
UNITS/ FORCE=NEWTON, MASS=KILOGRAM, LENGTH=METER, TIME=SECOND
!
! ---------------- Ground ----------------
PART/1, GROUND
MARKER/11, PART=1, QP=0,0,0
!
! ---------------- Free body (just a CG) ----------------
! Starts 10 m above ground (global Y). No joints -> 6 free DOF.
PART/2, QG=0,10,0, MASS=10.0, CM=21, IP=1,1,1
MARKER/21, PART=2, QP=0,0,0
!
! ---------------- Floating marker for the reaction ----------------
! VFORCE needs a "reaction" part; we use ground (nothing reacts there).
MARKER/12, PART=1, FLOATING
!
! ---------------- Gravity from user subroutine VFOSUB ----------------
! USER(mass, g): passed to Fortran as PAR(1), PAR(2).
! Force components are expressed in the ground marker 11 (global axes).
VFORCE/1, I=21, JFLOAT=12, RM=11, FUNCTION=USER(10.0, 9.81)
!
! ---------------- Output ----------------
REQUEST/1, TITLE=CG_height_and_velocity, F2=DY(21,11,11), F3=VY(21,11,11,11)
!
END


adams.acf

model
OUTPUT/REQSAVE, GRSAVE
SIMULATE/DYNAMIC, END=1.0, STEPS=100
STOP


usersub.f

C     Adams user subroutine: gravity force on the CG body.
C     Called by the VFORCE/1 statement via FUNCTION=USER(mass, g)
C        PAR(1) = mass [kg]
C        PAR(2) = g    [m/s^2]
C     RESULT(1:3) = force in the RM marker frame (global axes here)
C     RESULT(4:6) = torque (zero)
      SUBROUTINE VFOSUB(ID, TIME, PAR, NPAR, DFLAG, IFLAG, RESULT)
      IMPLICIT NONE
      INTEGER ID, NPAR
      DOUBLE PRECISION TIME, PAR(*), RESULT(6)
      LOGICAL DFLAG, IFLAG
C
      DOUBLE PRECISION MASS, G
C
      MASS = PAR(1)
      G    = PAR(2)
C
C     Gravity points along -Y
      RESULT(1) = 0.0D0
      RESULT(2) = -MASS*G
      RESULT(3) = 0.0D0
      RESULT(4) = 0.0D0
      RESULT(5) = 0.0D0
      RESULT(6) = 0.0D0
C
      RETURN
      END


usersub.f

#!/bin/bash
# Adams 2013 launcher (change if yours is called differently, e.g. "mdi")
ADAMS=${ADAMS:-adams2013}

# 1) compile the Fortran user subroutine into a shared library (usersub.so)
$ADAMS -c cr-u n usersub.f exit || exit 1

# 2) run the simulation using that library
$ADAMS -c ru-user i usersub.so model.acf exit
