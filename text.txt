function s = num2str8(x)

    % Number of characters before the decimal point
    intChars = length(sprintf('%.0f', abs(fix(x))));
    if x < 0
        intChars = intChars + 1;   % Account for minus sign
    end

    % Maximum decimals that fit in 8 characters
    decimals = max(1, 8 - intChars - 1);  % -1 for decimal point

    s = sprintf(['%8.' num2str(decimals) 'f'], x);

end
