function [yWin, uWin, TWin] = windowExperiment(y, u, nY, nU, numDelays)

    % ---------------------------------------------------------------
    % Convert everything to row vectors
    % ---------------------------------------------------------------

    y = double(y(:)).';
    u = double(u(:)).';


    % ---------------------------------------------------------------
    % Basic checks
    % ---------------------------------------------------------------

    if numel(y) ~= numel(u)
        error("U and Y must have the same number of samples.");
    end

    N = numel(y);

    if numDelays ~= max(nY,nU)
        numDelays = max(nY,nU);
    end

    if N <= numDelays
        error( ...
            "Experiment is too short. It contains %d samples, " + ...
            "but at least %d samples are required.", ...
            N, numDelays + 1);
    end


    % ---------------------------------------------------------------
    % Define the common prediction interval
    % ---------------------------------------------------------------
    %
    % We want the model to predict:
    %
    %       y(k)
    %
    % using information available before k.
    %
    % The first prediction is therefore:
    %
    %       k = numDelays + 1
    %
    % The last prediction is:
    %
    %       k = N
    %
    % Therefore every sequence has:
    %
    %       N - numDelays
    %
    % samples.
    %

    targetIndex = (numDelays + 1):N;

    TWin = y(targetIndex);


    % ---------------------------------------------------------------
    % Input sequences
    % ---------------------------------------------------------------
    %
    % Both inputs MUST have the same number of time samples because
    % they will be processed together by trainnet.
    %
    % We therefore provide the complete history available from the
    % common starting point.
    %
    % The convolution layers themselves determine how many samples
    % of that history are actually used.
    %

    inputIndex = numDelays:(N-1);

    yWin = y(inputIndex);
    uWin = u(inputIndex);


    % ---------------------------------------------------------------
    % Final consistency check
    % ---------------------------------------------------------------

    if numel(yWin) ~= numel(uWin)
        error( ...
            "Internal error: yWin and uWin have different lengths.");
    end

    if numel(yWin) ~= numel(TWin)
        error( ...
            "Internal error: input and target sequences have " + ...
            "different lengths. yWin=%d, uWin=%d, TWin=%d.", ...
            numel(yWin), ...
            numel(uWin), ...
            numel(TWin));
    end


    % ---------------------------------------------------------------
    % Final dimensions
    % ---------------------------------------------------------------
    %
    % All three are:
    %
    %       1 x (N-numDelays)
    %
    % which is the MATLAB sequence format:
    %
    %       features x time
    %

end
