function progressbar(i, N)

    persistent lastLength

    if i == 1
        lastLength = 0;
    end

    pct = 100*i/N;
    width = 40;
    n = round(width*i/N);

    str = sprintf('[%s%s] %6.2f%% (%d/%d)', ...
        repmat('=',1,n), repmat(' ',1,width-n), pct, i, N);

    % Erase previous line
    fprintf(repmat('\b',1,lastLength));

    % Print new line
    fprintf('%s', str);

    lastLength = length(str);

    if i == N
        fprintf('\n');
        clear progressbar
    end
end
