function T = read_adams_req(filename, varNames, matfile)
%READ_ADAMS_REQ  Read an Adams .req file into a MATLAB table.
%
%   T = read_adams_req('aircraft.req')
%   T = read_adams_req('aircraft.req', {'x','y','z','roll'})  % own names (no Time)
%   T = read_adams_req('aircraft.req', [], 'aircraft.mat')    % also save T
%
% File layout assumed:
%   line 3, first number = N  -> number of requests = number of header rows
%   lines 1..3+N              -> header (skipped)
%   remaining lines           -> numeric data: Time, then one column per output
%
% Result: one row per time step, one column per variable (first = Time).
% Nothing else is hard-coded, so adding/removing requests needs no change.

    if nargin < 2, varNames = []; end
    if nargin < 3, matfile  = []; end

    lines  = regexp(fileread(filename), '\r\n|\n|\r', 'split');
    numPat = '[-+]?(?:\d+\.?\d*|\.\d+)(?:[EeDd][-+]?\d+)?';

    if numel(lines) < 3
        error('read_adams_req:short', 'File has fewer than 3 lines.');
    end

    % --- header size: first number of line 3 --------------------------
    tok = regexp(lines{3}, numPat, 'match', 'once');
    if isempty(tok)
        error('read_adams_req:noCount', ...
              'No number found on line 3: "%s"', lines{3});
    end
    nReq = round(str2double(strrep(strrep(tok,'D','E'),'d','e')));
    skip = 3 + nReq;
    if numel(lines) <= skip
        error('read_adams_req:noData', 'No data after %d header lines.', skip);
    end
    header = lines(1:skip);

    % --- numeric rows ---------------------------------------------------
    rows = {};
    nIgnored = 0;
    for k = skip+1:numel(lines)
        s = lines{k};
        if isempty(strtrim(s)), continue; end
        v = parseNumericLine(s, numPat);
        if isempty(v)
            nIgnored = nIgnored + 1;
        else
            rows{end+1} = v; %#ok<AGROW>
        end
    end
    if isempty(rows)
        error('read_adams_req:noData', 'No numeric rows found after the header.');
    end

    nc   = numel(rows{1});
    good = cellfun(@numel, rows) == nc;
    nIgnored = nIgnored + nnz(~good);
    M = vertcat(rows{good});
    if nIgnored > 0
        warning('read_adams_req:ignored', ...
            '%d line(s) ignored (not numeric or different column count).', nIgnored);
    end

    % --- variable names ---------------------------------------------------
    if isempty(varNames)
        if nc - 1 == nReq            % one column per header row: use its text
            varNames = cell(1, nReq);
            for i = 1:nReq
                h = header{3 + i};
                t = regexp(h, 'TITLE\s*=\s*([^,]*)', 'tokens', 'once', 'ignorecase');
                if ~isempty(t), h = t{1}; end
                varNames{i} = matlab.lang.makeValidName(strtrim(h));
            end
        else
            varNames = arrayfun(@(i) sprintf('Var%d', i), 1:nc-1, 'UniformOutput', false);
        end
    end
    if numel(varNames) ~= nc - 1
        error('read_adams_req:names', ...
              '%d names given but the file has %d data columns after Time.', ...
              numel(varNames), nc - 1);
    end
    allNames = matlab.lang.makeUniqueStrings(['Time', varNames(:).'], {}, namelengthmax);

    T = array2table(M, 'VariableNames', allNames);
    T.Properties.UserData = struct('header', {header});

    fprintf('read_adams_req: %d requests in header, %d rows x %d columns.\n', ...
            nReq, size(T,1), size(T,2));

    if ~isempty(matfile)
        save(matfile, 'T');
        fprintf('Saved %s\n', matfile);
    end
end

function vals = parseNumericLine(s, numPat)
% Numbers on the line if it holds ONLY numbers/spaces/commas, else [].
% Handles Fortran D exponents and numbers glued by a minus sign.
    vals = [];
    tok = regexp(s, numPat, 'match');
    if isempty(tok), return; end
    rest = regexprep(s, numPat, '');
    if ~all(isspace(rest) | rest == ','), return; end
    vals = str2double(regexprep(tok, '[Dd]', 'E'));
    vals = vals(:).';
end
