% plot_stl.m
filename = 'model.stl';   % <-- your STL file

% Read STL (requires R2018b+; stlread returns a triangulation object)
TR = stlread(filename);

figure('Color','w');
trisurf(TR, 'FaceColor',[0.7 0.75 0.85], 'EdgeColor','none', ...
        'FaceLighting','gouraud', 'SpecularStrength',0.3);

axis equal; grid on;
xlabel('X'); ylabel('Y'); zlabel('Z');
view(3);
camlight headlight;
lighting gouraud;
title(filename, 'Interpreter','none');
rotate3d on;
