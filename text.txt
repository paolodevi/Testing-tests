A = model.A;
B = model.B;
nk = model.InputDelay;

na = length(A) - 1;
nu = length(B);

N = length(u(:,1));

y_pred = zeros(N,1);

for k = 1:N

    % AR part
    for j = 1:na
        if k-j >= 1
            y_pred(k) = y_pred(k) - A(j+1)*y_pred(k-j);
        end
    end

    % Input part
    for i = 1:nu

        nb = length(B{i});

        for j = 1:nb

            idx = k - nk(i) - j + 1;

            if idx >= 1
                y_pred(k) = y_pred(k) + B{i}(j)*u(idx,i);
            end

        end
    end
end
