#!/usr/bin/env python3

import argparse
import html
import re
from pathlib import Path


# ------------------------------------------------------------
# Fortran parser
# ------------------------------------------------------------

ROUTINE_RE = re.compile(
    r"^\s*(?:"
    r"program\s+(\w+)|"
    r"subroutine\s+(\w+)|"
    r"(?:recursive\s+|pure\s+|elemental\s+)?function\s+(\w+)"
    r")",
    re.IGNORECASE,
)

CALL_RE = re.compile(
    r"\bcall\s+([a-zA-Z_]\w*)",
    re.IGNORECASE,
)


def strip_comments(line):
    """
    Remove comments from a Fortran line.

    Handles:
      ! comments
      fixed-form comments beginning with C/c/* in column 1
    """

    if not line.strip():
        return ""

    # Fixed-form comment
    if line[0] in ("c", "C", "*"):
        return ""

    # Remove ! outside quoted strings
    result = []
    quote = None

    for char in line:
        if char in ("'", '"'):
            if quote is None:
                quote = char
            elif quote == char:
                quote = None

        if char == "!" and quote is None:
            break

        result.append(char)

    return "".join(result)


def parse_fortran_file(path):
    """
    Return:
        routines = {
            routine_name: {
                "name": ...,
                "file": ...,
                "calls": [...]
            }
        }
    """

    routines = {}

    try:
        text = path.read_text(errors="ignore")
    except Exception as e:
        print(f"Warning: cannot read {path}: {e}")
        return routines

    current = None

    for raw_line in text.splitlines():
        line = strip_comments(raw_line)

        if not line.strip():
            continue

        # Find routine definition
        match = ROUTINE_RE.match(line)

        if match:
            name = next(group for group in match.groups() if group)

            name = name.lower()

            current = {
                "name": name,
                "file": str(path),
                "calls": [],
            }

            routines[name] = current
            continue

        # Find CALL statements inside current routine
        if current:
            for call in CALL_RE.findall(line):
                call = call.lower()

                if call not in current["calls"]:
                    current["calls"].append(call)

    return routines


# ------------------------------------------------------------
# Build complete routine database
# ------------------------------------------------------------

def scan_source_tree(source_dir, extensions):
    database = {}

    for extension in extensions:
        for path in source_dir.rglob(f"*{extension}"):
            routines = parse_fortran_file(path)

            for name, routine in routines.items():

                # If duplicate routine names exist, keep the first
                # but warn the user.
                if name in database:
                    print(
                        f"Warning: duplicate routine '{name}':\n"
                        f"  {database[name]['file']}\n"
                        f"  {routine['file']}"
                    )
                else:
                    database[name] = routine

    return database


# ------------------------------------------------------------
# Find routines reachable from PALS
# ------------------------------------------------------------

def build_tree(root, database):

    visited = set()

    def recurse(name):

        # Prevent infinite recursion:
        #
        # A -> B -> C -> A
        #
        if name in visited:
            return {
                "name": name,
                "status": "recursive",
                "children": [],
            }

        if name not in database:
            return {
                "name": name,
                "status": "missing",
                "children": [],
            }

        visited.add(name)

        routine = database[name]

        children = []

        for call in routine["calls"]:
            children.append(recurse(call))

        return {
            "name": name,
            "file": routine["file"],
            "status": "found",
            "children": children,
        }

    return recurse(root.lower())


# ------------------------------------------------------------
# HTML generation
# ------------------------------------------------------------

def tree_to_html(node, node_id="node0"):

    name = html.escape(node["name"])

    status = node["status"]

    if status == "missing":
        css = "missing"
        extra = " <span class='missing-label'>[not found]</span>"

    elif status == "recursive":
        css = "recursive"
        extra = " <span class='recursive-label'>[recursive]</span>"

    else:
        css = "routine"
        extra = ""

    children = node["children"]

    if not children:
        return (
            f"<li class='{css}'>"
            f"<span class='name'>{name}</span>{extra}"
            f"</li>"
        )

    children_html = []

    for i, child in enumerate(children):
        child_id = f"{node_id}_{i}"
        children_html.append(tree_to_html(child, child_id))

    return f"""
    <li class="{css}">
        <span class="toggle"
              onclick="toggle('{node_id}')">▼</span>
        <span class="name">{name}</span>{extra}
        <ul id="{node_id}">
            {''.join(children_html)}
        </ul>
    </li>
    """


def generate_html(tree, output_file):

    tree_html = tree_to_html(tree)

    html_content = f"""<!DOCTYPE html>
<html lang="en">

<head>

<meta charset="UTF-8">

<title>Fortran Call Tree - PALS</title>

<style>

body {{
    font-family: Arial, sans-serif;
    margin: 30px;
    background: #fafafa;
    color: #222;
}}

h1 {{
    margin-bottom: 5px;
}}

.info {{
    color: #666;
    margin-bottom: 25px;
}}

.tree {{
    font-family: monospace;
    font-size: 14px;
}}

.tree ul {{
    list-style-type: none;
    margin: 4px 0;
    padding-left: 25px;
}}

.tree li {{
    margin: 4px 0;
}}

.toggle {{
    cursor: pointer;
    display: inline-block;
    width: 18px;
    user-select: none;
}}

.name {{
    cursor: default;
}}

.routine .name {{
    color: #1565c0;
}}

.missing .name {{
    color: #d32f2f;
}}

.recursive .name {{
    color: #ef6c00;
}}

.missing-label {{
    color: #d32f2f;
    font-size: 11px;
}}

.recursive-label {{
    color: #ef6c00;
    font-size: 11px;
}}

button {{
    margin-right: 8px;
    padding: 6px 12px;
    cursor: pointer;
}}

</style>

<script>

function toggle(id) {{
    const element = document.getElementById(id);

    if (element.style.display === "none") {{
        element.style.display = "block";
    }} else {{
        element.style.display = "none";
    }}
}}

function expandAll() {{
    document.querySelectorAll(".tree ul").forEach(
        element => element.style.display = "block"
    );
}}

function collapseAll() {{
    document.querySelectorAll(".tree ul").forEach(
        element => element.style.display = "none"
    );
}}

</script>

</head>

<body>

<h1>Fortran Call Tree</h1>

<div class="info">
Root routine: <b>PALS</b>
</div>

<button onclick="expandAll()">Expand all</button>
<button onclick="collapseAll()">Collapse all</button>

<div class="tree">

<ul>
{tree_html}
</ul>

</div>

</body>

</html>
"""

    output_file.write_text(html_content, encoding="utf-8")


# ------------------------------------------------------------
# Main
# ------------------------------------------------------------

def main():

    parser = argparse.ArgumentParser(
        description="Generate an interactive HTML call tree from Fortran source."
    )

    parser.add_argument(
        "source",
        type=Path,
        help="Root directory containing Fortran source files"
    )

    parser.add_argument(
        "--root",
        default="pals",
        help="Root routine (default: pals)"
    )

    parser.add_argument(
        "--output",
        default="fortran_call_tree.html",
        help="Output HTML file"
    )

    args = parser.parse_args()

    extensions = [
        ".f",
        ".f90",
        ".f95",
        ".f03",
        ".f08",
        ".for",
        ".F",
        ".F90",
    ]

    print(f"Scanning: {args.source}")

    database = scan_source_tree(
        args.source,
        extensions
    )

    print(f"Found {len(database)} routines.")

    root = args.root.lower()

    if root not in database:
        print(
            f"\nERROR: Could not find root routine '{root}'."
        )
        print(
            "Check the source directory and the routine name."
        )
        return

    print(f"Building call tree starting from '{root}'...")

    tree = build_tree(
        root,
        database
    )

    output = Path(args.output)

    generate_html(
        tree,
        output
    )

    print(f"\nDone!")
    print(f"HTML file: {output.resolve()}")


if __name__ == "__main__":
    main()
