      SUBROUTINE EUL2Q(ANGBOD, Q)
C     Converts Adams B313 Euler angles to a unit quaternion.
C     ANGBOD(1)=psi, ANGBOD(2)=theta, ANGBOD(3)=phi  [rad]
C     Q(1)=q0 (scalar), Q(2)=q1, Q(3)=q2, Q(4)=q3
C     Q describes the rotation body frame -> reference frame
      DOUBLE PRECISION ANGBOD(3), Q(4)
      DOUBLE PRECISION PSI, THE, PHI
      DOUBLE PRECISION CT, ST, SPLUS, SMINUS, QN

      PSI = ANGBOD(1)
      THE = ANGBOD(2)
      PHI = ANGBOD(3)

      CT     = DCOS(0.5D0*THE)
      ST     = DSIN(0.5D0*THE)
      SPLUS  = 0.5D0*(PSI+PHI)
      SMINUS = 0.5D0*(PSI-PHI)

      Q(1) = CT*DCOS(SPLUS)
      Q(2) = ST*DCOS(SMINUS)
      Q(3) = ST*DSIN(SMINUS)
      Q(4) = CT*DSIN(SPLUS)

C     Optional: normalise (guards against round-off) and keep q0 >= 0
      QN = DSQRT(Q(1)**2+Q(2)**2+Q(3)**2+Q(4)**2)
      IF (Q(1) .LT. 0.0D0) QN = -QN
      Q(1) = Q(1)/QN
      Q(2) = Q(2)/QN
      Q(3) = Q(3)/QN
      Q(4) = Q(4)/QN

      RETURN
      END
