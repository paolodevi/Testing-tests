function plotErrMatrix(data, threshold)
%PLOTERRMATRIX Matrix plot of data(i).(field).err
%   rows    = data(i)
%   columns = fields of data
%   cells with err > threshold (default 2) are red, others use parula.

if nargin < 2, threshold = 2; end

fn = fieldnames(data(1));
n  = numel(data);
m  = numel(fn);

% Collect err values (NaN where missing)
E = nan(n, m);
for i = 1:n
    for j = 1:m
        if isfield(data(i), fn{j}) && isstruct(data(i).(fn{j})) ...
                && isfield(data(i).(fn{j}), 'err') ...
                && isscalar(data(i).(fn{j}).err)
            E(i, j) = data(i).(fn{j}).err;
        end
    end
end

% Color range: parula spans [min, threshold]
lo = min(E(:), [], 'omitnan');
hi = threshold;
if isempty(lo) || isnan(lo) || lo >= hi, lo = hi - 1; end

figure;
imagesc(E);
colormap(parula);
caxis([lo hi]);
cb = colorbar;
cb.Label.String = 'err';
hold on;

% Red squares + text
for i = 1:n
    for j = 1:m
        v = E(i, j);
        if isnan(v), continue; end
        if v > threshold
            rectangle('Position', [j-0.5, i-0.5, 1, 1], ...
                      'FaceColor', 'r', 'EdgeColor', 'none');
            tc = 'w';
        elseif (v - lo) / (hi - lo) < 0.5
            tc = 'w';   % dark part of parula
        else
            tc = 'k';   % bright part of parula
        end
        text(j, i, sprintf('%.3g', v), 'HorizontalAlignment', 'center', ...
             'VerticalAlignment', 'middle', 'Color', tc, 'FontWeight', 'bold');
    end
end

% Grid lines between cells
for k = 0.5:1:m+0.5, plot([k k], [0.5 n+0.5], 'k-', 'LineWidth', 0.5); end
for k = 0.5:1:n+0.5, plot([0.5 m+0.5], [k k], 'k-', 'LineWidth', 0.5); end

% Axes
set(gca, 'XTick', 1:m, 'XTickLabel', fn, 'YTick', 1:n, ...
         'TickLabelInterpreter', 'none', 'XAxisLocation', 'top');
xtickangle(45);
ylabel('data index i');
axis tight; axis ij;
title(sprintf('err (red = err > %g)', threshold));
end
