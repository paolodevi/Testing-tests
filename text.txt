%% ------------------------------------------------------------------------
% Data
% -------------------------------------------------------------------------
Ts = mean(diff(t));      % Sampling time
Fs = 1/Ts;

alpha = detrend(alpha(:));
flap  = detrend(flap(:));
fore  = detrend(fore(:));

%% ------------------------------------------------------------------------
% Plot signals
% -------------------------------------------------------------------------
figure

subplot(3,1,1)
plot(t,alpha,'LineWidth',1.5)
grid on
ylabel('\alpha')

subplot(3,1,2)
plot(t,flap,'LineWidth',1.5)
grid on
ylabel('\delta_{flap}')

subplot(3,1,3)
plot(t,fore,'LineWidth',1.5)
grid on
ylabel('\delta_{fore}')
xlabel('Time [s]')

%% ------------------------------------------------------------------------
% Cross correlation
% -------------------------------------------------------------------------
figure

subplot(2,1,1)

[c,lags] = xcorr(flap,alpha,'coeff');

plot(lags*Ts,c,'LineWidth',1.5)
grid on
xlabel('Lag [s]')
ylabel('Correlation')
title('Flap vs Alpha')

[~,I] = max(c);

fprintf('\n');
fprintf('Maximum flap correlation = %.3f\n',c(I));
fprintf('Lag = %.4f s\n',lags(I)*Ts);

subplot(2,1,2)

[c,lags] = xcorr(fore,alpha,'coeff');

plot(lags*Ts,c,'LineWidth',1.5)
grid on
xlabel('Lag [s]')
ylabel('Correlation')
title('Foreplane vs Alpha')

[~,I] = max(c);

fprintf('Maximum foreplane correlation = %.3f\n',c(I));
fprintf('Lag = %.4f s\n',lags(I)*Ts);

%% ------------------------------------------------------------------------
% Estimate transfer functions
% -------------------------------------------------------------------------
u = iddata([],alpha,Ts);

y_flap = iddata(flap,alpha,Ts);
y_fore = iddata(fore,alpha,Ts);

% Second-order transfer functions
np = 2;
nz = 1;

G_flap = tfest(y_flap,np,nz);
G_fore = tfest(y_fore,np,nz);

disp('Estimated flap transfer function')
G_flap

disp('Estimated foreplane transfer function')
G_fore

%% ------------------------------------------------------------------------
% Compare prediction with measurements
% -------------------------------------------------------------------------
flap_est = lsim(G_flap,alpha,t);
fore_est = lsim(G_fore,alpha,t);

figure

subplot(2,1,1)

plot(t,flap,'b','LineWidth',1.5)
hold on
plot(t,flap_est,'r--','LineWidth',1.5)

grid on
legend('Measured','Estimated')
ylabel('\delta_{flap}')

subplot(2,1,2)

plot(t,fore,'b','LineWidth',1.5)
hold on
plot(t,fore_est,'r--','LineWidth',1.5)

grid on
legend('Measured','Estimated')
ylabel('\delta_{fore}')
xlabel('Time [s]')
