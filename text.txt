%% ---- User settings -------------------------------------------------
% Map the columns of your Sim array here
colT     = 1;    % time [s]  (set [] if you have no time column)
colX     = 2;    % x (north)
colY     = 3;    % y (east)
colZ     = 4;    % z
colRoll  = 5;
colPitch = 6;
colYaw   = 7;

anglesInDegrees = true;    % true if roll/pitch/yaw are in deg
zIsAltitude     = false;   % true if z is positive UP (it will be flipped to NED)
fps             = 30;      % animation frame rate
timeScaling     = 1;       % 1 = real time, 5 = 5x faster, etc.
recordVideo     = false;
videoFile       = 'flight.avi';

%% ---- Extract data --------------------------------------------------
if isempty(colT)
    dt = 0.01;                                  % set your sample time
    t = (0:size(Sim,1)-1)' * dt;
else
    t = Sim(:,colT);
end
x     = Sim(:,colX);
y     = Sim(:,colY);
z     = Sim(:,colZ);
roll  = Sim(:,colRoll);
pitch = Sim(:,colPitch);
yaw   = Sim(:,colYaw);

if anglesInDegrees
    roll = deg2rad(roll);  pitch = deg2rad(pitch);  yaw = deg2rad(yaw);
end
if zIsAltitude
    z = -z;                                     % NED: z positive down
end

%% ---- Resample to uniform time grid at animation fps ----------------
[t, iu] = unique(t);                            % remove duplicate times
x = x(iu); y = y(iu); z = z(iu);
roll = unwrap(roll(iu)); pitch = unwrap(pitch(iu)); yaw = unwrap(yaw(iu));

tu = (t(1):1/(fps*timeScaling):t(end))';
simdata = [tu, ...
           interp1(t,x,tu,'linear'), ...
           interp1(t,y,tu,'linear'), ...
           interp1(t,z,tu,'linear'), ...
           interp1(t,roll,tu,'linear'), ...
           interp1(t,pitch,tu,'linear'), ...
           interp1(t,yaw,tu,'linear')];
% Aero.Animation 6DoF array format: [time x y z phi theta psi]

%% ---- Animation object with built-in aircraft -----------------------
h = Aero.Animation;
h.FramesPerSecond = fps;
h.TimeScaling     = timeScaling;

h.createBody('pa24-250_orange.ac','Ac3d');      % built-in Piper PA-24 geometry
h.Bodies{1}.TimeSeriesSource     = simdata;
h.Bodies{1}.TimeSeriesSourceType = 'Array6DoF';

%% ---- Camera --------------------------------------------------------
h.Camera.PrefabPosition = 'Back';               % Back, Front, Left, Right, Top, Bottom
h.Camera.Offset         = [-60 0 -15];          % camera offset from body (body axes, m)

%% ---- Optional video recording --------------------------------------
if recordVideo
    h.VideoRecord      = 'on';
    h.VideoQuality     = 75;
    h.VideoCompression = 'Motion JPEG AVI';
    h.VideoFilename    = videoFile;
end

%% ---- Show and play -------------------------------------------------
h.show();
h.play();

if recordVideo
    h.VideoRecord = 'off';
end
