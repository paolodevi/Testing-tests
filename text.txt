
import h5py
import numpy as np


class H5Explorer:

    def __init__(self, filename):
        self.filename = filename
        self.file = None

    def open(self):
        self.file = h5py.File(self.filename, "r")
        print(f"\nOpened: {self.filename}")

    def close(self):
        if self.file is not None:
            self.file.close()
            print("File closed.")

    # --------------------------------------------------
    # Print complete HDF5 structure
    # --------------------------------------------------
    def show_structure(self):

        print(f"\nHDF5 STRUCTURE: {self.filename}\n")

        def visitor(name, obj):

            depth = name.count("/")
            indent = "    " * depth

            if isinstance(obj, h5py.Group):
                print(f"{indent}[GROUP] {name}")

            elif isinstance(obj, h5py.Dataset):

                print(
                    f"{indent}[DATASET] {name} | "
                    f"Shape: {obj.shape} | "
                    f"Type: {obj.dtype} | "
                    f"Size: {obj.size}"
                )

        self.file.visititems(visitor)

    # --------------------------------------------------
    # Display attributes
    # --------------------------------------------------
    def show_attributes(self, path="/"):

        obj = self.file[path]

        print(f"\nATTRIBUTES: {path}")

        if len(obj.attrs) == 0:
            print("No attributes found.")
            return

        for key, value in obj.attrs.items():
            print(f"{key}: {value}")

    # --------------------------------------------------
    # Preview dataset
    # --------------------------------------------------
    def preview(self, path, n=10):

        dataset = self.file[path]

        if not isinstance(dataset, h5py.Dataset):
            print("Selected object is not a dataset.")
            return

        print(f"\nDATASET: {path}")
        print(f"Shape: {dataset.shape}")
        print(f"Type: {dataset.dtype}")

        if dataset.ndim == 0:
            print(dataset[()])

        elif dataset.ndim == 1:
            print(dataset[:n])

        else:
            print(dataset[tuple(
                slice(0, min(s, n)) for s in dataset.shape
            )])

    # --------------------------------------------------
    # Load dataset into NumPy
    # --------------------------------------------------
    def load(self, path):

        dataset = self.file[path]

        if not isinstance(dataset, h5py.Dataset):
            raise TypeError(f"{path} is not a dataset.")

        return dataset[()]

    # --------------------------------------------------
    # Search datasets
    # --------------------------------------------------
    def search(self, keyword):

        print(f"\nSEARCH RESULTS: {keyword}")

        def visitor(name, obj):

            if isinstance(obj, h5py.Dataset):
                if keyword.lower() in name.lower():
                    print(f"{name} | {obj.shape} | {obj.dtype}")

        self.file.visititems(visitor)
