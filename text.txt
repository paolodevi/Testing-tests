%% NARX Network for SISO Time Series -- Trained Across Multiple Experiments
%
% Extends narx_siso.m to fit ONE NARX model using data from several
% separate experiments/runs (e.g. different test conditions), instead of
% a single long time series.
%
% Two things change vs. the single-experiment version:
%   1) Each experiment is windowed and kept as its OWN sequence -- never
%      concatenated end-to-end, which would create a fake "transition"
%      at the seam between two unrelated runs (the model would think the
%      last points of run A are the history leading into run B).
%   2) The train/val/test split happens at the EXPERIMENT level, not by
%      slicing time within one experiment. If the goal is a model that
%      generalizes across conditions, val/test need to be experiments
%      the model has NEVER seen -- not just a later time-slice of a run
%      it partially trained on.

%% 1) Load data: one cell array entry per experiment
% U, Y : Kx1 cell arrays, one cell per experiment. Each cell is a column
% vector of that experiment's u (or y) samples. Experiments can have
% different lengths -- that's fine, this doesn't require equal length.
%
%   U = {u_exp1; u_exp2; ...; u_expK};
%   Y = {y_exp1; y_exp2; ...; y_expK};

load("myMultiExpData.mat", "U", "Y")   % <-- EDIT: point at your own data
numExperiments = numel(U);

%% 2) Network architecture -- identical to the single-experiment version
nY = 2;
nU = 2;
numFilters = 15;

yBranch = [
    sequenceInputLayer(1, Name="yInput", MinLength=nY)
    convolution1dLayer(nY, numFilters, Name="convY")];

net = dlnetwork;
net = addLayers(net, yBranch);

uBranch = [
    sequenceInputLayer(1, Name="uInput", MinLength=nU)
    convolution1dLayer(nU, numFilters, Name="convU")
    concatenationLayer(1, 2, Name="cat")
    tanhLayer
    fullyConnectedLayer(1)];

net = addLayers(net, uBranch);
net = connectLayers(net, "convY", "cat/in2");

net = initialize(net);
net = dlupdate(@double, net);

%% 3) Split by EXPERIMENT, not by time
% Shuffle which experiments go where (fixed seed for reproducibility),
% then hold entire experiments out for val/test -- this is what actually
% tests whether the model generalizes to a case it hasn't seen, e.g. a
% mass/sink-rate combination that wasn't in the training set.
rng(0)
expOrder = randperm(numExperiments);

trainSplit = 0.6;
testSplit  = 0.2;   % remainder -> validation

numTrainExp = round(numExperiments * trainSplit);
numTestExp  = round(numExperiments * testSplit);

trainIdx = expOrder(1:numTrainExp);
testIdx  = expOrder(numTrainExp+1 : numTrainExp+numTestExp);
valIdx   = expOrder(numTrainExp+numTestExp+1 : end);

numDelays = max(nU, nY);   % every experiment must have at least this many samples

%% 4) Window each experiment independently, keep them as separate cells
% windowExperiment() (defined at the bottom of this file) applies the
% same delay-offsetting logic as the single-experiment script, but to
% one experiment at a time so no history leaks across boundaries.
[yTrainCell, uTrainCell, TTrainCell] = deal(cell(numTrainExp,1));
for k = 1:numTrainExp
    e = trainIdx(k);
    [yTrainCell{k}, uTrainCell{k}, TTrainCell{k}] = ...
        windowExperiment(Y{e}, U{e}, nY, nU, numDelays);
end

% --- Normalize using stats pooled across ALL training experiments ---
% One shared normalization for the whole model, fit only on training data.
allYTrain = vertcat(yTrainCell{:});
allUTrain = vertcat(uTrainCell{:});
muY = mean(allYTrain);  sigmaY = std(allYTrain);
muU = mean(allUTrain);  sigmaU = std(allUTrain);

normalize = @(c, mu, sigma) cellfun(@(x) (x-mu)/sigma, c, UniformOutput=false);
yTrainCell = normalize(yTrainCell, muY, sigmaY);
uTrainCell = normalize(uTrainCell, muU, sigmaU);
TTrainCell = normalize(TTrainCell, muY, sigmaY);

% A cell array -- one cell per experiment -- is read by arrayDatastore
% one cell (one full sequence) at a time by default, so each experiment
% becomes its own training observation instead of being glued together.
dsTrain = combine( ...
    arrayDatastore(yTrainCell), ...
    arrayDatastore(uTrainCell), ...
    arrayDatastore(TTrainCell));

% --- Validation experiments, same treatment ---
numValExp = numel(valIdx);
[yValCell, uValCell, TValCell] = deal(cell(numValExp,1));
for k = 1:numValExp
    e = valIdx(k);
    [yValCell{k}, uValCell{k}, TValCell{k}] = ...
        windowExperiment(Y{e}, U{e}, nY, nU, numDelays);
end
yValCell = normalize(yValCell, muY, sigmaY);
uValCell = normalize(uValCell, muU, sigmaU);
TValCell = normalize(TValCell, muY, sigmaY);

dsVal = combine( ...
    arrayDatastore(yValCell), ...
    arrayDatastore(uValCell), ...
    arrayDatastore(TValCell));

%% 5) Train
options = trainingOptions("lm", ...
    Plots="training-progress", ...
    Verbose=false, ...
    ValidationData=dsVal, ...
    MaxIterations=500, ...
    GradientTolerance=1e-8, ...
    StepTolerance=1e-8);

trainedNet = trainnet(dsTrain, net, "mse", options);

%% 6) Test: evaluate per held-out experiment, not pooled
% Reporting one RMSE per test experiment (rather than a single pooled
% number) shows whether performance is consistent across cases, or the
% model quietly fails on a subset of conditions that a pooled average
% would hide.
numTestExpActual = numel(testIdx);
rmseOpenPerExp = zeros(numTestExpActual,1);

for k = 1:numTestExpActual
    e = testIdx(k);
    [yTest, uTest, TTest] = windowExperiment(Y{e}, U{e}, nY, nU, numDelays);
    yTest = (yTest - muY) / sigmaY;
    uTest = (uTest - muU) / sigmaU;
    TTest = (TTest - muY) / sigmaY;

    YOpenLoop = predict(trainedNet, yTest, uTest);
    rmseOpenPerExp(k) = rmse(YOpenLoop, TTest);
end

resultsTable = table(testIdx', rmseOpenPerExp, ...
    VariableNames=["ExperimentIndex","RMSE_OpenLoop"])
fprintf("Mean open-loop RMSE across %d held-out experiments: %.4f\n", ...
    numTestExpActual, mean(rmseOpenPerExp));
fprintf("Worst-case open-loop RMSE: %.4f (experiment %d)\n", ...
    max(rmseOpenPerExp), testIdx(rmseOpenPerExp==max(rmseOpenPerExp)));

%% 7) (Optional) Choosing nY / nU / numFilters for the "ideal" model
% To find settings that generalize -- not just fit the training data --
% wrap steps 2-6 in a loop over candidate (nY, nU, numFilters) values and
% compare mean AND worst-case held-out RMSE across experiments for each
% candidate. Pick the smallest model whose worst-case error is still
% acceptable, rather than the one with the lowest mean (a low mean can
% hide one badly-fit case).
%
% With only a handful of experiments, consider k-fold cross-validation:
% rotate which experiments are held out as "test" across folds and
% average the result, instead of trusting a single random split.

%% Helper: consistent per-experiment windowing (same offsets as narx_siso.m)
function [yWin, uWin, TWin] = windowExperiment(y, u, nY, nU, numDelays)
    yWin = y(1+numDelays-nY : end-1);
    uWin = u(1+numDelays-nU : end-1);
    TWin = y(1+numDelays : end);
end
