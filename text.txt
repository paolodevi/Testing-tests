%% NARX Network for Single-Input / Single-Output (SISO) Time Series Forecasting
%
% Generic rewrite of the MathWorks MagLev NARX example, using a generic
% exogenous input u(t) and output y(t):
%
%   y(t) = f( y(t-1), y(t-2), ..., y(t-nY), u(t-1), u(t-2), ..., u(t-nU) )
%
% In words: the next output is predicted from a short window of its own
% recent past (the autoregressive, "AR" part) plus a window of recent
% input values (the "X" / exogenous part), passed through a nonlinear
% function f() that the network learns.
%
% Requires: Deep Learning Toolbox, Simulink (R2025a or later, for trainnet)

%% 1) Load data
% u : exogenous input,  column vector [numSamples x 1]
% y : output to predict, column vector [numSamples x 1]
% Replace this block with however you load your own data, e.g.:
%   data = readmatrix("mydata.csv");
%   u = data(:,1);
%   y = data(:,2);

load("myData.mat", "u", "y")   % <-- EDIT: point this at your own data

figure
tiledlayout(2,1)
nexttile
plot(u); xlabel("Time Step"); ylabel("u (input)")
nexttile
plot(y); xlabel("Time Step"); ylabel("y (output)")

%% 2) Define the NARX network architecture
nY = 2;           % how many past y-values feed back in (AR order)
nU = 2;           % how many past u-values are used (exogenous order)
numFilters = 15;  % width of the hidden layer: how many nonlinear
                  % features are learned from each delay window

% --- Branch 1: autoregressive (past y) ---
% sequenceInputLayer takes a scalar time series. convolution1dLayer with
% a kernel of length nY slides a window along the sequence: each output
% time step is a *learned* weighted sum of the nY most recent y-values.
% That's exactly the tapped-delay-line y(t-1)...y(t-nY) from the NARX
% equation above, computed for every t in one pass.
yBranch = [
    sequenceInputLayer(1, Name="yInput", MinLength=nY)
    convolution1dLayer(nY, numFilters, Name="convY")];

net = dlnetwork;
net = addLayers(net, yBranch);

% --- Branch 2: exogenous (past u), plus the layers that combine both ---
% Same tapped-delay idea, now on u(t-1)...u(t-nU). concatenationLayer
% merges the features from both branches, tanhLayer supplies the
% nonlinearity, and fullyConnectedLayer(1) maps everything down to the
% single predicted value of y(t). Together, cat -> tanh -> fc *is* the
% learned function f() in the NARX equation.
uBranch = [
    sequenceInputLayer(1, Name="uInput", MinLength=nU)
    convolution1dLayer(nU, numFilters, Name="convU")
    concatenationLayer(1, 2, Name="cat")
    tanhLayer
    fullyConnectedLayer(1)];

net = addLayers(net, uBranch);
net = connectLayers(net, "convY", "cat/in2");   % wire the y-branch into cat's 2nd input

% Initialize weights, then switch to double precision -- this small,
% tightly-fit regression network trains more accurately in double.
net = initialize(net);
net = dlupdate(@double, net);

%% 3) Split into train / test / validation (chronological -- never shuffle time series)
trainSplit = 0.6;   % first 60% of samples -> training
testSplit  = 0.2;   % next 20%  -> test (only evaluated after training is done)
% the remaining ~20% -> validation (used *during* training to watch for overfitting)

numSamples = numel(u);
numTrain = floor(numSamples * trainSplit);
numTest  = floor(numSamples * testSplit);
numVal   = numSamples - numTrain - numTest;

numDelays = max(nU, nY);   % the larger delay sets how many samples of
                            % "history" every split needs before its first prediction

% --- Training predictors/targets ---
% yTrain/uTrain are the raw windows the conv layers slide over. Targets
% (TTrain) start numDelays steps later, since the network needs that
% much history before it can make its first prediction.
yTrain = y(1+numDelays-nY : numTrain-1);
uTrain = u(1+numDelays-nU : numTrain-1);
TTrain = y(1+numDelays : numTrain);

% --- Normalize using TRAINING statistics only ---
% Normalizing speeds up/stabilizes gradient-based training. Using only
% the training set's mean/std (then reusing it on val/test) avoids
% leaking information from val/test into the fit.
muY = mean(yTrain);  sigmaY = std(yTrain);
muU = mean(uTrain);  sigmaU = std(uTrain);

uTrain = (uTrain - muU) / sigmaU;
yTrain = (yTrain - muY) / sigmaY;
TTrain = (TTrain - muY) / sigmaY;

% trainnet wants a single datastore combining both predictor branches + target
dsTrain = combine( ...
    arrayDatastore(yTrain, IterationDimension=2), ...
    arrayDatastore(uTrain, IterationDimension=2), ...
    arrayDatastore(TTrain, IterationDimension=2));

% --- Validation data (last block of the series) ---
yVal = y(numTrain+numTest+1+numDelays-nY : end-1);
uVal = u(numTrain+numTest+1+numDelays-nU : end-1);
TVal = y(numTrain+numTest+1+numDelays : end);

uVal = (uVal - muU) / sigmaU;
yVal = (yVal - muY) / sigmaY;
TVal = (TVal - muY) / sigmaY;

dsVal = combine( ...
    arrayDatastore(yVal, IterationDimension=2), ...
    arrayDatastore(uVal, IterationDimension=2), ...
    arrayDatastore(TVal, IterationDimension=2));

% --- Test data (middle block; untouched until after training) ---
yTest = y(numTrain+1+numDelays-nY : numTrain+numTest-1);
uTest = u(numTrain+1+numDelays-nU : numTrain+numTest-1);
TTest = y(numTrain+1+numDelays : numTrain+numTest);

uTest = (uTest - muU) / sigmaU;
yTest = (yTest - muY) / sigmaY;
TTest = (TTest - muY) / sigmaY;

%% 4) Train the network
options = trainingOptions("lm", ...   % Levenberg-Marquardt: good fit for
                                       % small nets/datasets in one batch
    Plots="training-progress", ...
    Verbose=false, ...
    ValidationData=dsVal, ...
    MaxIterations=500, ...
    GradientTolerance=1e-8, ...
    StepTolerance=1e-8);

trainedNet = trainnet(dsTrain, net, "mse", options);

%% 5) Test the network
% Open-loop forecasting: at every step, feed in the ACTUAL past y and u
% values (never the network's own predictions). Measures pure one-step-
% ahead accuracy.
YOpenLoop = predict(trainedNet, yTest, uTest);

% Closed-loop forecasting: feed the network's own previous predictions
% back in as if they were the real y history. This simulates the model
% running on its own several steps into the future -- a harder, more
% realistic test of how it would behave as a standalone forecaster.
YClosedLoop = yTest(1:nY);   % seed with the first nY known values

for idx = 1:numTest-numDelays
    YClosedLoop(end+1) = predict(trainedNet, ...
        YClosedLoop(end-nY+1:end), ...
        uTest(idx:idx+nU-1));
end
YClosedLoop = YClosedLoop(numDelays+1:end);   % drop the seed values

% --- Compare against targets ---
yTrainTest = (y(1:numTrain+numTest) - muY) / sigmaY;

figure
plot(yTrainTest); hold on
plot((1:numel(YOpenLoop))+numTrain+numDelays, YOpenLoop, "--")
plot((1:numel(YClosedLoop))+numTrain, YClosedLoop, "--")
xlabel("Time Step"); ylabel("y (normalized)")
title("NARX Network Predictions")
legend("Target","Open-Loop Prediction","Closed-Loop Prediction", ...
    Location="northwest")

rmseOpen   = rmse(YOpenLoop, TTest)
rmseClosed = rmse(YClosedLoop, TTest)

save("trainedNARXNet.mat", "trainedNet")
