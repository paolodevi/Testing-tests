#!/usr/bin/env python3
"""
Fortran call-tree generator (a small "doxygen for Fortran").

Usage examples
--------------
    python fortran_call_tree.py /path/to/src
    python fortran_call_tree.py /path/a /path/b /path/c --root pals --output tree.html
    python fortran_call_tree.py /path/src --ext .inc --ignore "mpi_*" --ignore dgemm
    python fortran_call_tree.py /path/src --expand-once --max-depth 12

Every folder is scanned recursively (symlinks followed). Extensions are
matched case-insensitively (.f .F .for .FOR .f90 .F90 ...).

In the HTML:
    p: program   s: subroutine   f: function   ?: not found in the sources
Click a routine name to see its inputs/outputs, description, file, callers
and callees.
"""

import argparse
import difflib
import fnmatch
import html
import json
import os
import re
import sys
import urllib.parse
from pathlib import Path

sys.setrecursionlimit(20000)

DEFAULT_EXTENSIONS = [".f", ".for", ".f77", ".ftn",
                      ".f90", ".f95", ".f03", ".f08", ".fpp"]
FIXED_FORM_EXTENSIONS = {".f", ".for", ".f77", ".ftn"}   # lowercase
DEFAULT_EXCLUDED_DIRS = {".git", ".svn", ".hg", "__pycache__"}

KIND_LETTER = {"program": "p", "subroutine": "s", "function": "f", "missing": "?"}


# ------------------------------------------------------------
# Regular expressions
# ------------------------------------------------------------

_TYPE = (
    r"(?:integer|real|double\s*precision|double\s*complex|complex|logical|"
    r"character|byte|type\s*\([^)]*\)|class\s*\([^)]*\))"
    r"(?:\s*\*\s*\d+|\s*\*\s*\(\s*[^)]*\)|\s*\([^)]*\))?"
)

# program x / subroutine x(...) / [real(8)|recursive|pure...] function x(...) [result(r)]
_ROUTINE_RE = re.compile(
    r"^\s*(?P<prefix>(?:(?:recursive|pure|elemental|impure|module|" + _TYPE + r")\s+)*)"
    r"(?P<kind>program|subroutine|function)\s+(?P<name>[a-z_]\w*)\s*"
    r"(?:\((?P<args>[^)]*)\))?(?P<rest>.*)$",
    re.IGNORECASE,
)
_RESULT_RE = re.compile(r"\bresult\s*\(\s*([a-z_]\w*)\s*\)", re.IGNORECASE)
_END_ROUTINE_RE = re.compile(
    r"^\s*end\s*(?:(?:program|subroutine|function)\b.*)?$", re.IGNORECASE)

_MODULE_RE = re.compile(
    r"^\s*module\s+(?!procedure\b|subroutine\b|function\b|pure\b|elemental\b|"
    r"recursive\b)([a-z_]\w*)\s*$", re.IGNORECASE)
_END_MODULE_RE = re.compile(r"^\s*end\s*module\b", re.IGNORECASE)

_INTERFACE_RE = re.compile(r"^\s*(?:abstract\s+)?interface\b", re.IGNORECASE)
_END_INTERFACE_RE = re.compile(r"^\s*end\s*interface\b", re.IGNORECASE)

_TYPE_DEF_RE = re.compile(r"^\s*type\b(?!\s*\()(?!\s+is\b)", re.IGNORECASE)
_END_TYPE_RE = re.compile(r"^\s*end\s*type\b", re.IGNORECASE)

# call foo(...)  /  call obj%method(...)
_CALL_RE = re.compile(r"\bcall\s+([a-z_]\w*(?:\s*%\s*[a-z_]\w*)*)", re.IGNORECASE)
# something( ... used to detect references to known functions
_FUNC_REF_RE = re.compile(r"\b([a-z_]\w*)\s*\(", re.IGNORECASE)

# declarations:  real(8), intent(in) :: a, b(3)      and      real a, b(3)
_DECL_COLON_RE = re.compile(
    r"^\s*(?P<type>" + _TYPE + r")(?P<attrs>(?:\s*,.*?)?)\s*::\s*(?P<ents>.+)$",
    re.IGNORECASE)
_DECL_OLD_RE = re.compile(
    r"^\s*(?P<type>" + _TYPE + r")\s+(?P<ents>[a-z_].*)$", re.IGNORECASE)

_INTENT_ATTR_RE = re.compile(r"intent\s*\(\s*(in\s*out|inout|in|out)\s*\)", re.IGNORECASE)
_DIM_ATTR_RE = re.compile(r"dimension\s*(\((?:[^()]|\([^()]*\))*\))", re.IGNORECASE)
_OPTIONAL_ATTR_RE = re.compile(r"\boptional\b", re.IGNORECASE)

_INTENT_STMT_RE = re.compile(
    r"^\s*intent\s*\(\s*(?P<i>in\s*out|inout|in|out)\s*\)\s*(?:::)?\s*(?P<ents>.+)$",
    re.IGNORECASE)
_DIM_STMT_RE = re.compile(r"^\s*dimension\s*(?:::)?\s*(?P<ents>.+)$", re.IGNORECASE)
_OPTIONAL_STMT_RE = re.compile(r"^\s*optional\s*(?:::)?\s*(?P<ents>.+)$", re.IGNORECASE)

_ENTITY_RE = re.compile(r"\s*([a-z_]\w*)\s*(\((?:[^()]|\([^()]*\))*\))?", re.IGNORECASE)

_NONEXEC_RE = re.compile(
    r"^\s*(?:use|implicit|external|intrinsic|common|dimension|data|parameter|save|"
    r"include|format|entry|equivalence|namelist|contains|optional|intent|public|"
    r"private|allocatable|pointer|target)\b", re.IGNORECASE)


# ------------------------------------------------------------
# Reading source: comments, continuation lines, fixed/free form
# ------------------------------------------------------------

def strip_inline(line):
    """Remove a trailing ! comment and blank out string contents."""
    out = []
    quote = None
    for ch in line:
        if quote:
            if ch == quote:
                quote = None
                out.append(ch)
            continue
        if ch in ("'", '"'):
            quote = ch
            out.append(ch)
            continue
        if ch == "!":
            break
        out.append(ch)
    return "".join(out)


def read_statements(path):
    """
    Return (raw_lines, is_fixed_form, statements).

    statements = [(first_line_idx, last_line_idx, lowercase_text), ...]
    with comments removed, continuation lines joined and ';' split.
    """
    text = path.read_text(errors="ignore")
    raw = text.splitlines()

    # .f files are fixed form, unless they clearly use F90 syntax ('::')
    fixed = path.suffix.lower() in FIXED_FORM_EXTENSIONS and "::" not in text

    logical = []          # [start, end, code]
    cur = None
    pending = False       # free form: previous line ended with '&'

    for i, line in enumerate(raw):
        if line.lstrip().startswith("#"):          # preprocessor
            continue

        if fixed:
            if line[:1] in ("c", "C", "*", "!") or not line.strip():
                continue
            is_cont = len(line) > 5 and line[5] not in " 0" and not line[:5].strip()
            code = strip_inline(line[6:] if is_cont else line)
            if not code.strip():
                continue
            if is_cont and cur is not None:
                cur[1] = i
                cur[2] += " " + code.strip()
            else:
                if cur is not None:
                    logical.append(cur)
                cur = [i, i, code]
        else:
            code = strip_inline(line)
            if not code.strip():
                continue
            next_pending = code.rstrip().endswith("&")
            if next_pending:
                code = code.rstrip()[:-1]
            if pending and cur is not None:
                piece = code.lstrip()
                if piece.startswith("&"):
                    piece = piece[1:]
                cur[1] = i
                cur[2] += " " + piece.strip()
            else:
                if cur is not None:
                    logical.append(cur)
                cur = [i, i, code]
            pending = next_pending

    if cur is not None:
        logical.append(cur)

    statements = []
    for start, end, code in logical:
        for part in code.split(";"):
            part = re.sub(r"^\s*\d+\s+", "", part.strip())   # drop numeric label
            if part:
                statements.append((start, end, part.lower()))

    return raw, fixed, statements


def comment_text(line, fixed):
    """Return the text of a comment line, or None if it is not a comment."""
    s = line.strip()
    if fixed and line[:1] in ("c", "C", "*"):
        return line[1:].strip()
    if s.startswith("!"):
        return s.lstrip("!").lstrip("<> ").strip()
    return None


def extract_doc(raw, start, end, fixed, limit=30):
    """Comment block just above the routine header, else just below it."""
    def collect(indices):
        lines = []
        for i in indices:
            t = comment_text(raw[i], fixed)
            if t is None:
                break
            lines.append(t)
        return lines

    above = collect(range(start - 1, -1, -1))[::-1]
    lines = above or collect(range(end + 1, len(raw)))
    lines = [l for l in lines if re.search(r"\w", l)]    # drop ----- separators
    return "\n".join(lines[:limit])


# ------------------------------------------------------------
# Fortran parser
# ------------------------------------------------------------

def split_top(text, sep=","):
    parts, depth, cur = [], 0, []
    for ch in text:
        if ch == "(":
            depth += 1
        elif ch == ")":
            depth -= 1
        if ch == sep and depth == 0:
            parts.append("".join(cur))
            cur = []
        else:
            cur.append(ch)
    parts.append("".join(cur))
    return [p.strip() for p in parts if p.strip()]


def norm_intent(text):
    t = re.sub(r"\s+", "", text.lower())
    return "inout" if t in ("inout",) else t


def record_decl(routine, type_str, attrs, ents):
    intent, optional, dims = None, False, ""
    if attrs:
        m = _INTENT_ATTR_RE.search(attrs)
        if m:
            intent = norm_intent(m.group(1))
        m = _DIM_ATTR_RE.search(attrs)
        if m:
            dims = m.group(1)
        optional = bool(_OPTIONAL_ATTR_RE.search(attrs))
    type_str = re.sub(r"\s+", " ", type_str.strip()) if type_str else ""

    for ent in split_top(ents):
        m = _ENTITY_RE.match(ent)
        if not m:
            continue
        d = routine["decls"].setdefault(m.group(1).lower(), {})
        if type_str:
            d["type"] = type_str
        if intent:
            d["intent"] = intent
        if optional:
            d["optional"] = True
        if m.group(2) or dims:
            d["dims"] = m.group(2) or dims


def record_attr_statement(routine, ents, intent=None, optional=False, dims=False):
    for ent in split_top(ents):
        m = _ENTITY_RE.match(ent)
        if not m:
            continue
        d = routine["decls"].setdefault(m.group(1).lower(), {})
        if intent:
            d["intent"] = intent
        if optional:
            d["optional"] = True
        if dims and m.group(2):
            d["dims"] = m.group(2)


def parse_fortran_file(path):
    """Return a list of routine dicts found in one file."""
    try:
        raw, fixed, statements = read_statements(path)
    except Exception as e:
        print(f"Warning: cannot read {path}: {e}")
        return []

    routines = []
    stack = []            # open routines (internal procedures nest)
    current = None
    module = ""
    iface_depth = 0
    in_type = False

    def close(routine, end_idx):
        routine["end_line"] = end_idx + 1
        routine["source"] = "\n".join(raw[routine["line"] - 1:end_idx + 1])

    for start, end, stmt in statements:

        # Skip interface blocks (they only declare, they don't define)
        if _INTERFACE_RE.match(stmt):
            iface_depth += 1
            continue
        if _END_INTERFACE_RE.match(stmt):
            iface_depth = max(0, iface_depth - 1)
            continue
        if iface_depth:
            continue

        # Skip derived-type definitions
        if _TYPE_DEF_RE.match(stmt):
            in_type = True
            continue
        if _END_TYPE_RE.match(stmt):
            in_type = False
            continue
        if in_type:
            continue

        m = _MODULE_RE.match(stmt)
        if m:
            module = m.group(1)
            continue
        if _END_MODULE_RE.match(stmt):
            module = ""
            while stack:
                close(stack.pop(), end)
            current = None
            continue

        # Routine header
        m = _ROUTINE_RE.match(stmt)
        if m:
            kind = m.group("kind").lower()
            name = m.group("name").lower()
            args = [a.strip() for a in (m.group("args") or "").split(",")
                    if a.strip() and a.strip() != "*"]
            prefix_type = re.sub(
                r"\b(recursive|pure|elemental|impure|module)\b", "",
                m.group("prefix"), flags=re.IGNORECASE)
            prefix_type = re.sub(r"\s+", " ", prefix_type).strip()
            rm = _RESULT_RE.search(m.group("rest") or "")
            current = {
                "name": name,
                "kind": kind,
                "file": str(path),
                "line": start + 1,
                "module": module,
                "args": args,
                "result": (rm.group(1).lower() if rm else name) if kind == "function" else "",
                "prefix_type": prefix_type,
                "decls": {},
                "body": [],
                "calls": [],
                "doc": extract_doc(raw, start, end, fixed),
                "fixed": fixed,
            }
            routines.append(current)
            stack.append(current)
            continue

        if _END_ROUTINE_RE.match(stmt):
            if stack:
                close(stack.pop(), end)
            current = stack[-1] if stack else None
            continue

        if current is None:
            continue

        # Declarations
        m = _DECL_COLON_RE.match(stmt) or _DECL_OLD_RE.match(stmt)
        if m:
            record_decl(current, m.group("type"), m.groupdict().get("attrs"), m.group("ents"))
            continue
        m = _INTENT_STMT_RE.match(stmt)
        if m:
            record_attr_statement(current, m.group("ents"), intent=norm_intent(m.group("i")))
            continue
        m = _OPTIONAL_STMT_RE.match(stmt)
        if m:
            record_attr_statement(current, m.group("ents"), optional=True)
            continue
        m = _DIM_STMT_RE.match(stmt)
        if m:
            record_attr_statement(current, m.group("ents"), dims=True)
            continue
        if _NONEXEC_RE.match(stmt):
            continue

        # Executable statement
        current["body"].append(stmt)
        for call in _CALL_RE.findall(stmt):
            callee = call.split("%")[-1].strip()
            if callee not in current["calls"]:
                current["calls"].append(callee)

    while stack:                               # file ended without END
        close(stack.pop(), len(raw) - 1)

    return routines


# ------------------------------------------------------------
# Build complete routine database
# ------------------------------------------------------------

def iter_source_files(sources, extensions, excluded_dirs):
    exts = {(e if e.startswith(".") else "." + e).lower() for e in extensions}
    seen_dirs = set()

    for source in sources:
        if source.is_file():
            yield source
            continue

        for dirpath, dirnames, filenames in os.walk(source, followlinks=True):
            real = os.path.realpath(dirpath)
            if real in seen_dirs:               # symlink loops / overlapping sources
                dirnames[:] = []
                continue
            seen_dirs.add(real)

            dirnames[:] = sorted(d for d in dirnames if d not in excluded_dirs)
            for filename in sorted(filenames):
                path = Path(dirpath) / filename
                if path.suffix.lower() in exts:
                    yield path


def scan_source_tree(sources, extensions, excluded_dirs):
    database = {}
    duplicates = []
    nfiles = 0

    for path in iter_source_files(sources, extensions, excluded_dirs):
        nfiles += 1
        for routine in parse_fortran_file(path):
            name = routine["name"]
            if name in database:
                duplicates.append((name, database[name]["file"], routine["file"]))
            else:
                database[name] = routine

    return database, duplicates, nfiles


def resolve_function_references(database):
    """
    Functions are used as  y = f(x)  without CALL, so add them to the calls of
    every routine that references a known function name.
    """
    functions = {n for n, r in database.items() if r["kind"] == "function"}
    if not functions:
        return

    for routine in database.values():
        for stmt in routine["body"]:
            for ident in _FUNC_REF_RE.findall(stmt):
                ident = ident.lower()
                if ident not in functions or ident == routine["name"]:
                    continue
                if routine["decls"].get(ident, {}).get("dims"):    # it is an array
                    continue
                if ident not in routine["calls"]:
                    routine["calls"].append(ident)


def apply_ignore(database, patterns):
    if not patterns:
        return
    patterns = [p.lower() for p in patterns]
    for routine in database.values():
        routine["calls"] = [
            c for c in routine["calls"]
            if not any(fnmatch.fnmatch(c, p) for p in patterns)
        ]


# ------------------------------------------------------------
# Call tree
# ------------------------------------------------------------

def build_tree(root, database, max_depth=0, expand_once=False):
    reachable = set()
    expanded = set()
    stack = []
    counter = [0]

    def recurse(name, depth):
        counter[0] += 1
        reachable.add(name)

        if name not in database:
            return {"name": name, "kind": "missing", "status": "missing", "children": []}

        routine = database[name]
        base = {"name": name, "kind": routine["kind"]}

        if name in stack:                         # real recursion: A -> B -> A
            return {**base, "status": "recursive", "children": []}
        if expand_once and name in expanded:
            return {**base, "status": "repeated", "children": []}
        if max_depth and depth >= max_depth and routine["calls"]:
            return {**base, "status": "truncated", "children": []}

        expanded.add(name)
        stack.append(name)
        children = [recurse(c, depth + 1) for c in routine["calls"]]
        stack.pop()

        return {**base, "status": "found", "children": children}

    tree = recurse(root, 0)
    return tree, reachable, counter[0]


# ------------------------------------------------------------
# Inputs / outputs
# ------------------------------------------------------------

def infer_intent(arg, body):
    """
    Heuristic for old code without INTENT:
      out?  assigned inside the routine
      ?     passed to another routine (unknown)
      in?   only read
    """
    a = re.escape(arg)
    assign = re.compile(
        rf"(?:^\s*|\)\s*|\bdo\s+(?:\d+\s*,?\s*)?){a}\s*"
        rf"(?:\((?:[^()]|\([^()]*\))*\))?\s*=(?!=)")
    passed = re.compile(
        rf"\bcall\s+[a-z_]\w*(?:\s*%\s*[a-z_]\w*)*\s*\(.*\b{a}\b")

    is_passed = False
    for stmt in body:
        if assign.search(stmt):
            return "out?"
        if passed.search(stmt):
            is_passed = True
    return "?" if is_passed else "in?"


def classify_args(routine):
    inputs, outputs, unknown = [], [], []

    for arg in routine["args"]:
        d = routine["decls"].get(arg, {})
        intent = d.get("intent") or infer_intent(arg, routine["body"])
        entry = {
            "name": arg,
            "type": d.get("type", ""),
            "dims": d.get("dims", ""),
            "intent": intent,
            "optional": bool(d.get("optional")),
        }
        if intent in ("in", "in?"):
            inputs.append(entry)
        elif intent in ("out", "out?"):
            outputs.append(entry)
        elif intent == "inout":
            inputs.append(entry)
            outputs.append(entry)
        else:
            unknown.append(entry)

    if routine["kind"] == "function":
        res = routine["result"]
        d = routine["decls"].get(res, {})
        outputs.insert(0, {
            "name": res,
            "type": routine["prefix_type"] or d.get("type", ""),
            "dims": d.get("dims", ""),
            "intent": "return",
            "optional": False,
        })

    return inputs, outputs, unknown


def build_info(database, reachable, include_source=True):
    callers = {n: [] for n in reachable}
    for name in reachable:
        routine = database.get(name)
        if not routine:
            continue
        for callee in routine["calls"]:
            if callee in callers and name not in callers[callee]:
                callers[callee].append(name)

    info = {}
    for name in reachable:
        routine = database.get(name)
        if routine is None:
            info[name] = {"name": name, "kind": "missing",
                          "calls": [], "callers": sorted(callers[name])}
            continue
        inputs, outputs, unknown = classify_args(routine)
        path = Path(routine["file"]).resolve()
        info[name] = {
            "abs": str(path),
            "uri": path.as_uri(),
            "vscode": "vscode://file/" + urllib.parse.quote(path.as_posix().lstrip("/"), safe="/:")
                      + f":{routine['line']}",
            "end_line": routine["end_line"],
            "fixed": routine["fixed"],
            "source": routine["source"] if include_source else "",
            "name": name,
            "kind": routine["kind"],
            "file": routine["file"],
            "line": routine["line"],
            "module": routine["module"],
            "doc": routine["doc"],
            "inputs": inputs,
            "outputs": outputs,
            "unknown": unknown,
            "calls": routine["calls"],
            "callers": sorted(callers[name]),
        }
    return info

STATUS_LABELS = {
    "missing": ("not found", "missing-label"),
    "recursive": ("recursive", "recursive-label"),
    "repeated": ("expanded above", "repeated-label"),
    "truncated": ("max depth reached", "repeated-label"),
}


def tree_to_html(node):
    status = node["status"]
    kind = node["kind"]
    name = html.escape(node["name"])

    label = ""
    if status in STATUS_LABELS:
        text, css = STATUS_LABELS[status]
        label = f" <span class='{css}'>[{text}]</span>"

    head = (f"<span class='name' data-name='{name}'>"
            f"<span class='kind'>{KIND_LETTER[kind]}:</span> {name}</span>{label}")

    classes = [kind] + ([status] if status != "found" and status != kind else [])
    cls = " ".join(classes)

    if not node["children"]:
        return f"<li class='{cls}'><span class='spacer'></span>{head}</li>"

    kids = "".join(tree_to_html(c) for c in node["children"])
    return f"<li class='{cls}'><span class='toggle'></span>{head}<ul>{kids}</ul></li>"

def generate_html(tree, info, root, stats, output_file):
    page = (HTML_TEMPLATE
            .replace("__ROOT__", html.escape(root))
            .replace("__STATS__", html.escape(stats))
            .replace("__TREE__", tree_to_html(tree))
            .replace("__INFO__", json.dumps(info).replace("</", "<\\/")))
    output_file.write_text(page, encoding="utf-8")

def main():
    parser = argparse.ArgumentParser(
        description="Generate an interactive HTML call tree from Fortran source.")

    parser.add_argument(
        "sources", nargs="+", type=Path,
        help="One or more folders (scanned recursively) or files with the Fortran sources")
    parser.add_argument("--root", default="pals", help="Root routine (default: pals)")
    parser.add_argument("--output", default="fortran_call_tree.html", help="Output HTML file")
    parser.add_argument("--ext", action="append", default=[],
                        help="Extra file extension to scan, e.g. --ext .inc (repeatable)")
    parser.add_argument("--exclude", action="append", default=[],
                        help="Folder name to skip, e.g. --exclude build (repeatable)")
    parser.add_argument("--ignore", action="append", default=[],
                        help="Routine name/pattern to leave out of the tree, "
                             "e.g. --ignore 'mpi_*' (repeatable)")
    parser.add_argument("--max-depth", type=int, default=0,
                        help="Limit the depth of the tree (0 = unlimited)")
    parser.add_argument("--expand-once", action="store_true",
                        help="Expand each routine only the first time it appears "
                             "(much smaller HTML for big code bases)")
    parser.add_argument("--no-source", action="store_true",
                        help="Do not embed routine source code in the HTML (smaller file)")
    parser.add_argument("-v", "--verbose", action="store_true",
                        help="Print every duplicate / missing routine")

    args = parser.parse_args()

    sources = []
    for source in args.sources:
        if not source.exists():
            print(f"ERROR: '{source}' does not exist.")
            return 1
        sources.append(source)

    extensions = DEFAULT_EXTENSIONS + args.ext
    excluded = DEFAULT_EXCLUDED_DIRS | set(args.exclude)

    print("Scanning:")
    for source in sources:
        print(f"  {source}")

    database, duplicates, nfiles = scan_source_tree(sources, extensions, excluded)
    print(f"Scanned {nfiles} files, found {len(database)} routines.")

    if duplicates:
        print(f"Warning: {len(duplicates)} duplicate routine definition(s); "
              f"the first one found is used.")
        shown = duplicates if args.verbose else duplicates[:10]
        for name, first, other in shown:
            print(f"  {name}:\n    {first}\n    {other}")
        if len(shown) < len(duplicates):
            print("  ... (use -v to list all)")

    resolve_function_references(database)
    apply_ignore(database, args.ignore)

    root = args.root.lower()
    if root not in database:
        print(f"\nERROR: Could not find root routine '{root}'.")
        close = difflib.get_close_matches(root, list(database), n=5)
        if close:
            print("Did you mean: " + ", ".join(close))
        print("Check the source folders and the routine name.")
        return 1

    print(f"Building call tree starting from '{root}'...")
    tree, reachable, nodes = build_tree(root, database, args.max_depth, args.expand_once)
    info = build_info(database, reachable, include_source=not args.no_source)

    missing = sorted(n for n in reachable if n not in database)
    found = len(reachable) - len(missing)
    if missing:
        print(f"\n{len(missing)} called routine(s) not found in the sources "
              f"(library, intrinsic, dummy procedure or missing folder?):")
        shown = missing if args.verbose else missing[:40]
        print("  " + ", ".join(shown) + (" ..." if len(shown) < len(missing) else ""))

    if nodes > 100000 and not args.expand_once:
        print(f"\nNote: the tree has {nodes} nodes; consider --expand-once or --max-depth.")

    stats = f"{found} routines found, {len(missing)} not found, {nodes} tree nodes"
    output = Path(args.output)
    generate_html(tree, info, root, stats, output)

    size_mb = output.stat().st_size / 1e6
    print(f"\nDone!")
    print(f"HTML file: {output.resolve()} ({size_mb:.1f} MB)")
    if size_mb > 50:
        print("Note: the file is large; use --no-source, --expand-once or --max-depth to shrink it.")
    return 0


if __name__ == "__main__":
    sys.exit(main())
